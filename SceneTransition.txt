local populatedInventories = {}

local function populateInventory(inventoryFrame, categoryName)
	local scrollingFrame = inventoryFrame:WaitForChild("ScrollingFrame")
	for _, child in ipairs(scrollingFrame:GetChildren()) do
		if child:GetAttribute("Item") == true then
			child:Destroy()
		end
	end

	local items = GetInventoryItems:InvokeServer(categoryName)
	if not items then
		warn("Не удалось загрузить инвентарь '" .. categoryName .. "'")
		return
	end

	for _, itemData in ipairs(items) do
		local itemClone = itemTemplate:Clone()
		itemClone:SetAttribute("Item", true)
		itemClone.Name = itemData.name .. "_Frame"
		local button = itemClone:FindFirstChild("ImageButton")
		if button then
			button.Activated:Connect(function()
				local equippedName = currentEquipped[categoryName]
				if equippedName == itemData.name then
					EquipItemEvent:FireServer(categoryName, nil)
					print("Снятие: " .. itemData.name .. " из " .. categoryName)
				else
					EquipItemEvent:FireServer(categoryName, itemData.name)
					print("Экипировка: " .. itemData.name .. " из " .. categoryName)
				end
			end)
		end
		itemClone.Parent = scrollingFrame
	end
	populatedInventories[inventoryFrame] = true
end

local function openColorPicker(palette, categoryName, parentInventory)
	if currentOpenInventory == palette then
		if colorPickers[palette] then
			colorPickers[palette]:Destroy()
			colorPickers[palette] = nil
		end
		palette.Visible = false
		currentOpenInventory = nil
		print("Палитра закрыта: " .. palette.Name)
		return
	end

	for _, pal in ipairs({ColorEyes, ColorFacialHair, ColorHair, ColorSkin, ColorRace}) do
		if pal.Visible and pal ~= palette then
			if colorPickers[pal] then
				colorPickers[pal]:Destroy()
				colorPickers[pal] = nil
			end
			pal.Visible = false
		end
	end

	if parentInventory then
		parentInventory.Visible = true
	end

	palette.Visible = true
	currentOpenInventory = palette

	local initialColor = currentCategoryColors[categoryName] or Color3.new(1,1,1)

	local function onColorChanged(newColor)
		currentCategoryColors[categoryName] = newColor
		ChangePartColorEvent:FireServer(categoryName, newColor)
		print("Предпросмотр " .. categoryName .. ": " .. tostring(newColor))
	end
	
	local function onClosed()
		palette.Visible = false
		if currentOpenInventory == palette then
			currentOpenInventory = nil
		end
		if colorPickers[palette] == picker then
			colorPickers[palette] = nil
		end
		if categoryName == "Skin" then
			skinColorFrame.BackgroundColor3 = currentCategoryColors.Skin
		end
		print("Пикер закрыт: " .. categoryName)
	end

	local picker = ColorPickerModule.new(palette, initialColor, onColorChanged, onClosed)
	colorPickers[palette] = picker
end

local function switchToMainInventory(targetFrame)
	if currentOpenInventory == targetFrame then
		closeAllInventories()
		return
	end

	for pal, picker in pairs(colorPickers) do
		picker:Destroy()
		colorPickers[pal] = nil
	end
	for frame in pairs(allFrames) do
		frame.Visible = false
	end

	targetFrame.Visible = true
	currentOpenInventory = targetFrame
	print("Открыт: " .. targetFrame.Name)

	if not populatedInventories[targetFrame] and inventoryMapping[targetFrame] then
		populateInventory(targetFrame, inventoryMapping[targetFrame])
	end
end

local categoryItemCache = {}

local function preloadCategories()
	-- ИСПРАВЛЕНО: добавлена "Race" для корректной работы рандомайзера
	local cats = {"Shirt", "Pants", "Extra", "Eyes", "Hair", "FacialHair", "Mouth", "Race"}
	for _, cat in ipairs(cats) do
		categoryItemCache[cat] = GetInventoryItems:InvokeServer(cat)
	end
end
preloadCategories()

randomButton.MouseButton1Click:Connect(function()
	closeAllInventories()
	local rng = Random.new()

	-- Обязательные категории (всегда новые предметы)
	local required = {"Shirt", "Pants", "Eyes", "Mouth", "Race"}
	for _, cat in ipairs(required) do
		local items = categoryItemCache[cat]
		if items and #items > 0 then
			local chosen = items[rng:NextInteger(1, #items)]
			EquipItemEvent:FireServer(cat, chosen.name)
		end
	end

	-- Опциональные категории с вероятностями и возможностью снятия
	local optional = {
		{name = "Hair",       chance = 0.85},
		{name = "FacialHair", chance = 0.20},
		{name = "Extra",      chance = 0.20},
	}
	for _, opt in ipairs(optional) do
		local items = categoryItemCache[opt.name]
		if items and #items > 0 and rng:NextNumber() < opt.chance then
			local chosen = items[rng:NextInteger(1, #items)]
			EquipItemEvent:FireServer(opt.name, chosen.name)
		else
			EquipItemEvent:FireServer(opt.name, nil)
		end
	end

	-- Применяем цвета
	local colorable = {"Eyes", "Hair", "FacialHair", "Race"}
	for _, cat in ipairs(colorable) do
		local col = getPleasantColor()
		currentCategoryColors[cat] = col
		ChangePartColorEvent:FireServer(cat, col)
	end

	-- Цвет кожи отдельно (телесная палитра)
	local skinCol = skinColors[rng:NextInteger(1, #skinColors)]
	currentCategoryColors.Skin = skinCol
	ChangePartColorEvent:FireServer("Skin", skinCol)
	skinColorFrame.BackgroundColor3 = skinCol

	print("?? Рандомайзер выполнен!")
end)

local function returnToMenu()
	if isTransitioning then return end
	closeAllInventories()
	isTransitioning = true

	TweenService:Create(mainContainer, TweenInfo.new(0.8, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
		{Position = originalMainPos}):Play()

	local hidePos = UDim2.new(0.5, 0, 1.2, 0)
	local tweenBottom = TweenService:Create(buttonContainer, TweenInfo.new(0.8, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
		{Position = hidePos})
	tweenBottom:Play()
	tweenBottom.Completed:Connect(function()
		buttonContainer.Visible = false
	end)

	local tweenInfo = TweenInfo.new(0.8, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
	TweenService:Create(customizeLeft, tweenInfo, {Position = UDim2.new(-0.1, 0, 0.5, 0)}):Play()
	TweenService:Create(customizeRight, tweenInfo, {Position = UDim2.new(1.1, 0, 0.5, 0)}):Play()
	TweenService:Create(customizeText, tweenInfo, {Position = UDim2.new(0.5, 0, -0.1, 0)}):Play()
	local lastTween = TweenService:Create(leftRight, tweenInfo, {Position = UDim2.new(0.5, 0, -0.1, 0)})
	lastTween:Play()

	local randomTween = TweenService:Create(randomFrame, tweenInfo, {Position = UDim2.new(1.1, 0, 0.15, 0)})
	randomTween:Play()
	randomTween.Completed:Connect(function()
		randomFrame.Visible = false
	end)
	lastTween.Completed:Connect(function()
		customizeLeft.Visible = false
		customizeRight.Visible = false
		customizeText.Visible = false
		leftRight.Visible = false
		isTransitioning = false
	end)

	TweenService:Create(bgLeft, tweenInfo, {Position = bgLeftOriginalPos}):Play()
	TweenService:Create(bgRight, tweenInfo, {Position = bgRightOriginalPos}):Play()

	local camera = workspace.CurrentCamera
	if not camera then return end
	local angleVal = Instance.new("NumberValue")
	angleVal.Value = -math.pi/2
	local tweenCam = TweenService:Create(angleVal, TweenInfo.new(1.0, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
		{Value = 0})
	angleVal.Changed:Connect(function()
		camera.CFrame = CFrame.new(cameraBasePos) * CFrame.Angles(0, angleVal.Value, 0)
	end)
	tweenCam.Completed:Connect(function() angleVal:Destroy() end)
	tweenCam:Play()
end

	local ACTIVE_COLOR = Color3.fromRGB(180, 180, 180)
	local MALE_INACTIVE_COLOR = Color3.fromRGB(62, 127, 206)
	local FEMALE_INACTIVE_COLOR = Color3.fromRGB(194, 124, 206)

	local maleButton = genderFrame:WaitForChild("Frame"):WaitForChild("MaleButton")
	local femaleButton = genderFrame:WaitForChild("Frame"):WaitForChild("FemaleButton")

	local function updateGenderButtons(selectedGender)
		if selectedGender == "male" then
			maleButton.BackgroundColor3 = ACTIVE_COLOR
			femaleButton.BackgroundColor3 = FEMALE_INACTIVE_COLOR
		elseif selectedGender == "female" then
			maleButton.BackgroundColor3 = MALE_INACTIVE_COLOR
			femaleButton.BackgroundColor3 = ACTIVE_COLOR
		end
	end

	-- Первичная подсветка из StringValue
	local currentGender = getCurrentGender()
	if currentGender then
		updateGenderButtons(currentGender)
	end

	-- Слушаем изменения пола (если он меняется не только по кнопке, но и с сервера)
	genderValue.Changed:Connect(function(newValue)
		updateGenderButtons(newValue)
	end)

	-- Функция запроса смены пола
	local function applyGender(gender)
		if gender == "male" or gender == "female" then
			changeGenderEvent:FireServer(gender)
			-- Мгновенно подсвечиваем кнопку для отзывчивости интерфейса
			updateGenderButtons(gender)
			print("Запрос на смену пола отправлен: " .. gender)
		else
			warn("Неизвестный пол: " .. tostring(gender))
		end
	end

	maleButton.MouseButton1Click:Connect(function() applyGender("male") end)
	femaleButton.MouseButton1Click:Connect(function() applyGender("female") end)
end

-- Рекурсивный поиск кнопки по имени внутри elementMenu
local function findButton(parent, name)
	for _, child in ipairs(parent:GetChildren()) do
		if child.Name == name and child:IsA("GuiButton") then
			return child
		end
		local found = findButton(child, name)
		if found then return found end
	end
	return nil
end

local currentInfoFrame = nil   -- какой фрейм открыт сейчас

-- Вспомогательные функции анимации
local function closeInfoFrame(infoFrame, callback)
	local tween = TweenService:Create(infoFrame,
		TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
		{Position = INFO_HIDE_POS})
	tween:Play()
	tween.Completed:Connect(function()
		infoFrame.Visible = false
		if callback then callback() end
	end)
end

local function showInfoFrame(infoFrame, callback)
	infoFrame.Visible = true
	infoFrame.Position = INFO_HIDE_POS
	local tween = TweenService:Create(infoFrame,
		TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
		{Position = INFO_SHOW_POS})
	tween:Play()
	if callback then
		tween.Completed:Connect(callback)
	end
end

-- Обработчик нажатия на кнопку стихии
local function onElementButtonClicked(infoFrame)
	if isInfoAnimating then return end

	if currentInfoFrame == infoFrame then
		isInfoAnimating = true
		closeInfoFrame(infoFrame, function()
			currentInfoFrame = nil
			isInfoAnimating = false
		end)
	else
		isInfoAnimating = true
		local previous = currentInfoFrame
		local function openNew()
			showInfoFrame(infoFrame, function()
				currentInfoFrame = infoFrame
				isInfoAnimating = false
			end)
		end
		if previous then
			closeInfoFrame(previous, openNew)
		else
			openNew()
		end
	end
end

if earthButton then earthButton.MouseButton1Click:Connect(function() onElementButtonClicked(earthInfoFrame) end) end

if gloomButton then gloomButton.MouseButton1Click:Connect(function() onElementButtonClicked(gloomInfoFrame) end) end

-- Финальный переход из меню стихий (выбор сделан)
local function transitionToFinal(chosenElement)
	if isTransitioning then return end
	isTransitioning = true

	-- Закрываем все инвентари/палитры
	closeAllInventories()

	-- === Скрываем ElementMenuGui ===
	local tweenInfo = TweenInfo.new(0.8, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)

	TweenService:Create(elementButtonContainer, tweenInfo,
		{Position = UDim2.new(0.5, 0, 1.2, 0)}):Play()

	TweenService:Create(elementBackFrameUp, tweenInfo,
		{Position = UDim2.new(0.5, 0, -0.1, 0)}):Play()
	TweenService:Create(elementBackFrameDown, tweenInfo,
		{Position = UDim2.new(0.5, 0, 1.1, 0)}):Play()
	local backFrameTween = TweenService:Create(elementBackFrame, tweenInfo,
		{Position = UDim2.new(-0.1, 0, 0.4, 0)})
	backFrameTween:Play()
	backFrameTween.Completed:Connect(function()
		elementBackFrame.Visible = false
		elementMenu.Enabled = false
	end)

	TweenService:Create(yourChoiceFrame, tweenInfo,
		{Position = UDim2.new(0.5, 0, 1.1, 0)}):Play()

	if currentInfoFrame then
		local frameToClose = currentInfoFrame
		currentInfoFrame = nil
		TweenService:Create(frameToClose,
			TweenInfo.new(0.5, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
			{Position = INFO_HIDE_POS}):Play()
	end

	-- === Фоновые панели MainMenuGui ===
	TweenService:Create(bgLeft, tweenInfo,
		{Position = UDim2.new(-2, 0, 0.5, 0)}):Play()
	TweenService:Create(bgRight, tweenInfo,
		{Position = UDim2.new(2, 0, 0.5, 0)}):Play()

	-- === Подготовка EntranceGui (будет появляться позже) ===
	local entranceGui = playerGui:WaitForChild("EntranceGui")
	local entranceBackFrame = entranceGui:WaitForChild("BackFrame")
	local entranceFrame = entranceGui:WaitForChild("Frame")
	entranceGui.Enabled = true
	entranceBackFrame.Visible = true
	entranceBackFrame.BackgroundTransparency = 1
	entranceFrame.Visible = true
	entranceFrame.BackgroundTransparency = 1

	-- === Поворот и подъём камеры ===
	local camera = workspace.CurrentCamera
	if not camera then
		isTransitioning = false
		return
	end

	camera.CameraType = Enum.CameraType.Scriptable

	local angleVal = Instance.new("NumberValue")
	angleVal.Value = -math.pi
	local targetAngle = -3 * math.pi / 2

	local heightVal = Instance.new("NumberValue")
	heightVal.Value = 0
	local targetHeight = 3   -- подъём на 3 студита

	local function updateCamera()
		camera.CFrame = CFrame.new(cameraBasePos + Vector3.new(0, heightVal.Value, 0))
			* CFrame.Angles(0, angleVal.Value, 0)
	end

	angleVal.Changed:Connect(updateCamera)
	heightVal.Changed:Connect(updateCamera)

	local tweenAngle = TweenService:Create(angleVal,
		TweenInfo.new(2.0, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
		{Value = targetAngle})
	local tweenHeight = TweenService:Create(heightVal,
		TweenInfo.new(2.0, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
		{Value = targetHeight})

	tweenAngle:Play()
	tweenHeight:Play()

	tweenAngle.Completed:Connect(function()
		-- Пауза после поворота/подъёма
		task.wait(0.5)

		-- Движение вперёд
		local forwardDistance = 30
		local targetCFrame = camera.CFrame * CFrame.new(0, 0, -forwardDistance)

		local tweenForward = TweenService:Create(camera,
			TweenInfo.new(1.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out),
			{CFrame = targetCFrame})
		tweenForward:Play()

		-- Примерно в конце движения (через 0.6 секунды) начинаем проявлять EntranceGui
		task.spawn(function()
			task.wait(0.6)
			local appearInfo = TweenInfo.new(0.6, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
			local tweenBack = TweenService:Create(entranceBackFrame, appearInfo, {BackgroundTransparency = 0})
			local tweenFrame = TweenService:Create(entranceFrame, appearInfo, {BackgroundTransparency = 0})
			tweenBack:Play()
			tweenFrame:Play()
		end)

		tweenForward.Completed:Connect(function()
			angleVal:Destroy()
			heightVal:Destroy()
			isTransitioning = false
			print("Финальный переход завершён. Элемент:", chosenElement)
		end)
		
		task.wait(0.5) -- Небольшая пауза после анимации

		-- Отправляем запрос на сервер вместе с выбором игрока
		local requestTeleport = ReplicatedStorage:WaitForChild("RequestTeleport")
		requestTeleport:FireServer(chosenElement) -- chosenElement - это имя выбранной стихии
	end)
end

ElementChosenEvent.Event:Connect(function(elementName)
	print("Выбрана стихия:", elementName, "– выполняю финальный переход.")
	transitionToFinal(elementName)
end)
